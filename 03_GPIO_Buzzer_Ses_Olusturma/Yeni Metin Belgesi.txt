#include "main.h"
void SystemClock_Config(void);
static void MX_GPIO_Init(void);

int main(void) {
  HAL_Init(); SystemClock_Config(); MX_GPIO_Init();
  while (1) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_3, GPIO_PIN_SET);   // Buzzer Ötüyor
    HAL_Delay(500);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_3, GPIO_PIN_RESET); // Buzzer Sustu
    HAL_Delay(500);
  }
}
