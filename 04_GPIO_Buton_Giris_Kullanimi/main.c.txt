#include "main.h"
void SystemClock_Config(void);
static void MX_GPIO_Init(void);

int main(void) {
  HAL_Init(); SystemClock_Config(); MX_GPIO_Init();
  uint8_t buton_durumu = 0;
  
  while (1) {
    // Pin giriş (Input) olarak ayarlandı ve okunuyor
    buton_durumu = HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0);
    HAL_Delay(50);
  }
}
