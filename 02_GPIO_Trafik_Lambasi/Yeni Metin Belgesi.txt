#include "main.h"
void SystemClock_Config(void);
static void MX_GPIO_Init(void);

int main(void) {
  HAL_Init(); SystemClock_Config(); MX_GPIO_Init();
  while (1) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_SET);   // Kırmızı Yak
    HAL_Delay(3000);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_RESET); // Kırmızı Söndür
    
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_1, GPIO_PIN_SET);   // Sarı Yak
    HAL_Delay(1000);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_1, GPIO_PIN_RESET); // Sarı Söndür
    
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_2, GPIO_PIN_SET);   // Yeşil Yak
    HAL_Delay(3000);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_2, GPIO_PIN_RESET); // Yeşil Söndür
  }
}
